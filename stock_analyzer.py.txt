import yfinance as yf
import pandas_ta as ta
import pandas as pd

def get_signal(df):
    if df is None or len(df) < 21:
        return "INSUFFICIENT DATA"

    # Indicators (as per your Pine Script)
    df['EMA9'] = ta.ema(df['Close'], length=9)
    df['EMA21'] = ta.ema(df['Close'], length=21)
    df['RSI'] = ta.rsi(df['Close'], length=14)
    df['AvgVol'] = ta.sma(df['Volume'], length=20)

    last = df.iloc[-1]
    prev = df.iloc[-2]

    # Logic
    bull_cross = (prev['EMA9'] <= prev['EMA21']) and (last['EMA9'] > last['EMA21'])
    bear_cross = (prev['EMA9'] >= prev['EMA21']) and (last['EMA9'] < last['EMA21'])
    vol_filter = last['Volume'] > last['AvgVol']

    if bull_cross and last['RSI'] > 50 and vol_filter:
        return "STRONG BUY ✅"
    elif bear_cross and last['RSI'] < 50 and vol_filter:
        return "STRONG SELL ❌"
    else:
        return "NEUTRAL ⚪"

def run_dashboard():
    # User Input
    print("\n--- GPT-5 STOCK ANALYZER ---")
    user_input = input("Enter stock symbols (e.g., NVDA, AAPL, TSLA): ")
    tickers = [t.strip().upper() for t in user_input.split(',')]

    results = []

    for symbol in tickers:
        print(f"Analyzing {symbol}...")
        try:
            # Get 1 Hour data
            data_1h = yf.download(symbol, interval="1h", period="2y", progress=False)
            # Get 1 Day data
            data_1d = yf.download(symbol, interval="1d", period="2y", progress=False)

            score_1h = get_signal(data_1h)
            score_1d = get_signal(data_1d)

            results.append({
                "Symbol": symbol,
                "1H Signal": score_1h,
                "Daily Signal": score_1d
            })
        except Exception as e:
            results.append({"Symbol": symbol, "1H Signal": "ERROR", "Daily Signal": "ERROR"})

    # Display Results
    print("\n" + "="*45)
    print(f"{'SYMBOL':<10} | {'1 HOUR SCORE':<15} | {'DAILY SCORE':<15}")
    print("-" * 45)
    for res in results:
        print(f"{res['Symbol']:<10} | {res['1H Signal']:<15} | {res['Daily Signal']:<15}")
    print("="*45)

if __name__ == "__main__":
    run_dashboard()
